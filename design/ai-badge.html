// src/components/AiBadge.tsx
interface Props {
  /** ai=✨AI（默认） | remote=☁联网 | thinking=🕐会先思考 */
  kind?: "ai" | "remote" | "thinking";
  /** xs 给 v6.0 卡片动作条用 */
  size?: "sm" | "xs";
}

// 调用方不自己判断，避免有人漏判：
<AiBadge kind={t.group === "ai" ? "ai" : "remote"} />
